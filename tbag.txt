#!/bin/bash
# =====================================================================
#  TBAG - Tablet Baglama Araci  (GitHub'dan calisan surum)
#  Pardus akilli tahtalara adb/scrcpy ile tablet baglamak icin.
# =====================================================================

set -u

renk()  { printf "\n\033[1;36m==> %s\033[0m\n" "$1"; }
ok()    { printf "\033[1;32m  [OK] %s\033[0m\n" "$1"; }
uyari() { printf "\033[1;33m  [!]  %s\033[0m\n" "$1"; }
hata()  { printf "\033[1;31m  [HATA] %s\033[0m\n" "$1"; }


# ---------------------------------------------------------------
# 0) GEREKLI KOMUTLAR VAR MI
# ---------------------------------------------------------------
if ! command -v adb >/dev/null 2>&1; then
    hata "adb bulunamadi. Once kurulum scriptini calistirdiginizdan emin olun."
    exit 1
fi
if ! command -v scrcpy >/dev/null 2>&1; then
    hata "scrcpy bulunamadi. Once kurulum scriptini calistirdiginizdan emin olun."
    exit 1
fi

# ---------------------------------------------------------------
# IP formatini basitce dogrula
# ---------------------------------------------------------------
ip_dogrula() {
    local ip="$1"
    if [[ "$ip" =~ ^([0-9]{1,3}\.){3}[0-9]{1,3}$ ]]; then
        return 0
    fi
    return 1
}
port_dogrula() {
    local p="$1"
    if [[ "$p" =~ ^[0-9]{2,5}$ ]] && [ "$p" -ge 1 ] && [ "$p" -le 65535 ]; then
        return 0
    fi
    return 1
}

oku_ip() {
    local prompt="$1" ip=""
    while true; do
        read -r -p "$prompt" ip
        if ip_dogrula "$ip"; then
            echo "$ip"
            return 0
        fi
        uyari "Gecersiz IP adresi, tekrar deneyin. (Ornek: 192.168.1.25)" >&2
    done
}
oku_port() {
    local prompt="$1" p=""
    while true; do
        read -r -p "$prompt" p
        if port_dogrula "$p"; then
            echo "$p"
            return 0
        fi
        uyari "Gecersiz port, tekrar deneyin. (Ornek: 5555)" >&2
    done
}

# ---------------------------------------------------------------
# 1) ONCEKI ADB BAGLANTILARINI TEMIZLE
# ---------------------------------------------------------------
renk "Onceki adb baglantilari kapatiliyor"
adb disconnect >/dev/null 2>&1
ok "adb disconnect calistirildi"

# ---------------------------------------------------------------
# 2) BAGLANTI TURU SECIMI
# ---------------------------------------------------------------
renk "Baglanti turunu secin"
echo "test kodu 123"
echo "  1) Kablosuz (Wi-Fi)"
echo "  2) Kablolu (USB uzerinden IP ile)"
BAGLANTI_TURU=""
while true; do
    read -r -p "Seciminiz [1-2]: " BAGLANTI_TURU
    case "$BAGLANTI_TURU" in
        1|2) break ;;
        *) uyari "Lutfen 1 veya 2 girin." ;;
    esac
done

# ---------------------------------------------------------------
# 3) EKRAN ORANI SECIMI
# ---------------------------------------------------------------
renk "Goruntu ekran oranini secin"
echo "  1) Tabletin kendi ekran oranini koru"
echo "  2) 16:9'a donustur (akilli tahtaya tam sigmasi icin)"
ORAN_SECIMI=""
while true; do
    read -r -p "Seciminiz [1-2]: " ORAN_SECIMI
    case "$ORAN_SECIMI" in
        1|2) break ;;
        *) uyari "Lutfen 1 veya 2 girin." ;;
    esac
done

# ---------------------------------------------------------------
# 4) BAGLANTI BILGILERI VE ADB BAGLANTISI
# ---------------------------------------------------------------
SERIAL=""

adb_baglan_dogrula() {
    # $1 = serial (ip:port)
    local s="$1" i durum
    for i in 1 2 3 4 5; do
        durum="$(adb devices | awk -v s="$s" '$1==s {print $2}')"
        if [ "$durum" = "device" ]; then
            return 0
        fi
        sleep 1
    done
    return 1
}

if [ "$BAGLANTI_TURU" = "2" ]; then
    # ---- KABLOLU: IP + PORT -> adb connect ----
    renk "Kablolu baglanti bilgileri"
    IP=$(oku_ip "Tabletin IP adresi: ")
    PORT=$(oku_port "Port numarasi: ")
    SERIAL="${IP}:${PORT}"

    echo "  Baglaniliyor: adb connect $SERIAL"
    BAGLANTI_CIKTI="$(adb connect "$SERIAL" 2>&1)"
    echo "  $BAGLANTI_CIKTI"

    if ! adb_baglan_dogrula "$SERIAL"; then
        hata "Cihaza baglanilamadi. IP/port bilgilerini ve ayni ag uzerinde oldugunuzu kontrol edin."
        exit 1
    fi
    ok "Baglanti basarili: $SERIAL"

else
    # ---- KABLOSUZ: IP, pair port, pair code, sonra connect port ----
    renk "Kablosuz eslesme (pairing) bilgileri"
    echo "  (Tabletinizde Ayarlar > Gelistirici Secenekleri > Kablosuz Hata Ayiklama"
    echo "   bolumunden 'Eslesme Kodu ile Eslestir' secenegini acin.)"
    IP=$(oku_ip "Tabletin IP adresi: ")
    PAIR_PORT=$(oku_port "Eslesme (pairing) portu: ")
    read -r -p "Eslesme kodu (6 haneli): " PAIR_CODE

    echo "  Eslesiliyor: adb pair $IP:$PAIR_PORT"
    ESLESME_BASARILI=0
    for deneme in 1 2 3; do
        ESLESME_CIKTI="$(adb pair "${IP}:${PAIR_PORT}" "$PAIR_CODE" 2>&1)"
        echo "  $ESLESME_CIKTI"
        if echo "$ESLESME_CIKTI" | grep -qi "Successfully paired"; then
            ESLESME_BASARILI=1
            break
        fi
        uyari "Eslesme basarisiz oldu (deneme $deneme/3)."
        if [ "$deneme" -lt 3 ]; then
            read -r -p "Eslesme kodunu tekrar girin: " PAIR_CODE
        fi
    done

    if [ "$ESLESME_BASARILI" -ne 1 ]; then
        hata "Eslesme basarisiz. Eslesme portu/kodu dogru mu ve ayni ag uzerinde misiniz kontrol edin."
        exit 1
    fi
    ok "Eslesme basarili"

    CONNECT_PORT=$(oku_port "Baglanti (connect) portu: ")
    SERIAL="${IP}:${CONNECT_PORT}"

    echo "  Baglaniliyor: adb connect $SERIAL"
    BAGLANTI_CIKTI="$(adb connect "$SERIAL" 2>&1)"
    echo "  $BAGLANTI_CIKTI"

    if ! adb_baglan_dogrula "$SERIAL"; then
        hata "Cihaza baglanilamadi. Baglanti portunu kontrol edin."
        exit 1
    fi
    ok "Baglanti basarili: $SERIAL"
fi

# ---------------------------------------------------------------
# 5) EKRAN ORANI HESAPLAMASI (16:9 secildiyse)
# ---------------------------------------------------------------
CROP_PARAM=()

if [ "$ORAN_SECIMI" = "2" ]; then
    renk "Tablet ekran cozunurlugu okunuyor"
    SIZE_LINE="$(adb -s "$SERIAL" shell wm size 2>/dev/null | tail -n1)"
    RES="$(echo "$SIZE_LINE" | grep -oE '[0-9]+x[0-9]+')"

    if [ -z "$RES" ]; then
        uyari "Cozunurluk okunamadi, 16:9 kirpma uygulanmadan devam edilecek."
    else
        WIDTH="${RES%x*}"
        HEIGHT="${RES#*x}"
        echo "  Algilanan cozunurluk: ${WIDTH}x${HEIGHT}"

        # Yatay mi dikey mi oldugunu belirle, hedef oranin
        # hangi eksende kirpilmasi gerektigini hesapla.
        if [ "$WIDTH" -ge "$HEIGHT" ]; then
            # Yatay tablet -> hedef 16:9
            ORAN_ASIM=$(awk -v w="$WIDTH" -v h="$HEIGHT" 'BEGIN{ print (w/h > 16/9) ? 1 : 0 }')
            if [ "$ORAN_ASIM" -eq 1 ]; then
                TARGET_H="$HEIGHT"
                TARGET_W=$(awk -v h="$HEIGHT" 'BEGIN{ printf "%d", h*16/9 }')
                [ "$TARGET_W" -gt "$WIDTH" ] && TARGET_W="$WIDTH"
                OFFSET_X=$(( (WIDTH - TARGET_W) / 2 ))
                OFFSET_Y=0
            else
                TARGET_W="$WIDTH"
                TARGET_H=$(awk -v w="$WIDTH" 'BEGIN{ printf "%d", w*9/16 }')
                [ "$TARGET_H" -gt "$HEIGHT" ] && TARGET_H="$HEIGHT"
                OFFSET_Y=$(( (HEIGHT - TARGET_H) / 2 ))
                OFFSET_X=0
            fi
        else
            # Dikey tablet -> hedef 9:16 (ayni oranin dikey karsiligi)
            ORAN_ASIM=$(awk -v w="$WIDTH" -v h="$HEIGHT" 'BEGIN{ print (h/w > 16/9) ? 1 : 0 }')
            if [ "$ORAN_ASIM" -eq 1 ]; then
                TARGET_W="$WIDTH"
                TARGET_H=$(awk -v w="$WIDTH" 'BEGIN{ printf "%d", w*16/9 }')
                [ "$TARGET_H" -gt "$HEIGHT" ] && TARGET_H="$HEIGHT"
                OFFSET_Y=$(( (HEIGHT - TARGET_H) / 2 ))
                OFFSET_X=0
            else
                TARGET_H="$HEIGHT"
                TARGET_W=$(awk -v h="$HEIGHT" 'BEGIN{ printf "%d", h*9/16 }')
                [ "$TARGET_W" -gt "$WIDTH" ] && TARGET_W="$WIDTH"
                OFFSET_X=$(( (WIDTH - TARGET_W) / 2 ))
                OFFSET_Y=0
            fi
        fi

        # Cift sayiya yuvarla (video kodlayici uyumlulugu icin)
        TARGET_W=$(( TARGET_W - (TARGET_W % 2) ))
        TARGET_H=$(( TARGET_H - (TARGET_H % 2) ))
        OFFSET_X=$(( OFFSET_X - (OFFSET_X % 2) ))
        OFFSET_Y=$(( OFFSET_Y - (OFFSET_Y % 2) ))

        CROP_VALUE="${TARGET_W}:${TARGET_H}:${OFFSET_X}:${OFFSET_Y}"
        CROP_PARAM=(--crop "$CROP_VALUE")
        ok "16:9 kirpma uygulanacak: $CROP_VALUE (kaynak: ${WIDTH}x${HEIGHT})"
    fi
else
    ok "Tabletin kendi ekran orani korunacak"
fi

# ---------------------------------------------------------------
# 6) SCRCPY BASLAT
# ---------------------------------------------------------------
renk "scrcpy baslatiliyor"
echo "  Cikmak icin scrcpy penceresini kapatin veya Ctrl+C basin."
sleep 1

scrcpy -s "$SERIAL" --max-size 1280 --video-bit-rate 2M --max-fps 30 -f "${CROP_PARAM[@]}"
SCRCPY_DURUM=$?

if [ "$SCRCPY_DURUM" -ne 0 ]; then
    hata "scrcpy beklenmedik sekilde kapandi (kod: $SCRCPY_DURUM)."
    exit "$SCRCPY_DURUM"
fi

exit 0
