#!/bin/bash
# =====================================================================
#  Android Cihaz Bağlama Araç Kutusu
#  Kablolu baglantiyi otomatik algilar, TCP/IP moduna alip baglanir,
#  ardindan grafik arayuz ile ekran oranini sorup scrcpy baslatir.
# =====================================================================

set -u

# Tek bir oturumun calisip calismadigini kontrol et (flock)
LOCK="/tmp/android-cihaz-baglama.lock"
exec 200>"$LOCK"
flock -n 200 || exit 0

# scrcpy zaten aciksa tekrar tetikleme yapma
pgrep -x scrcpy >/dev/null 2>&1 && exit 0

# ---------------------------------------------------------------
# Active X/Grafik Oturumunu Tespiti (Udev veya Arka Plan Tetiklemeleri Icin)
# ---------------------------------------------------------------
if [ -z "${DISPLAY:-}" ]; then
    GRAFIK_KULLANICI="$(who | awk '/\(:0(\.0)?\)|:0$/ {print $1; exit}')"
    if [ -z "$GRAFIK_KULLANICI" ]; then
        GRAFIK_KULLANICI="$(loginctl list-sessions --no-legend 2>/dev/null | awk '{print $3; exit}')"
    fi
    
    if [ -n "$GRAFIK_KULLANICI" ]; then
        export DISPLAY=":0"
        export XAUTHORITY="/home/${GRAFIK_KULLANICI}/.Xauthority"
    fi
fi

UYGULAMA_ADI="Android Cihaz Bağlama"

zenity_mesaj() {
    local tur="$1" metin="$2"
    zenity --"$tur" --title="$UYGULAMA_ADI" --text="$metin" --width=350 2>/dev/null || true
}

# ---------------------------------------------------------------
# 1) ADB VE USB CIHAZ KONTROLU
# ---------------------------------------------------------------
# Cihazin tam algilanmasi icin kisa bekleme
sleep 2

adb disconnect >/dev/null 2>&1

USB_SERIAL=""
for deneme in $(seq 1 15); do
    USB_SERIAL="$(adb devices | awk '$2=="device" && $1 !~ /:/ {print $1; exit}')"
    [ -n "$USB_SERIAL" ] && break
    sleep 1
done

if [ -z "$USB_SERIAL" ]; then
    # USB uzerinden bagli cihaz yoksa sessizce veya uyari ile cik
    exit 0
fi

# ---------------------------------------------------------------
# 2) IP TESPITI, TCP/IP MODUNA ALMA VE BAGLANMA (KABLO CIKARILABILIR HALE GELIR)
# ---------------------------------------------------------------
TCPIP_PORT=5555
IP="$(adb -s "$USB_SERIAL" shell ip route show dev wlan0 2>/dev/null \
        | grep -oE 'src [0-9]{1,3}(\.[0-9]{1,3}){3}' | awk '{print $2}' | head -n1)"

if [ -z "$IP" ]; then
    IP="$(adb -s "$USB_SERIAL" shell ip -f inet addr show wlan0 2>/dev/null \
            | grep -oE 'inet [0-9]{1,3}(\.[0-9]{1,3}){3}' | awk '{print $2}' | head -n1)"
fi

if [ -z "$IP" ]; then
    zenity_mesaj "error" "Tabletin Wi-Fi IP adresi okunamadı. Tabletin Wi-Fi ağında olduğundan emin olun."
    exit 1
fi

# Cihazi TCP/IP moduna al ve IP uzerinden baglan
adb -s "$USB_SERIAL" tcpip "$TCPIP_PORT" >/dev/null 2>&1
sleep 2

SERIAL="${IP}:${TCPIP_PORT}"
adb connect "$SERIAL" >/dev/null 2>&1

# Baglantinin kuruldugunu dogrula (5 sn kadar dene)
BAGLANDI=0
for i in 1 2 3 4 5; do
    DURUM="$(adb devices | awk -v s="$SERIAL" '$1==s {print $2}')"
    if [ "$DURUM" = "device" ]; then
        BAGLANDI=1
        break
    fi
    sleep 1
done

if [ "$BAGLANDI" -ne 1 ]; then
    zenity_mesaj "error" "Kablosuz bağlantı kurulamadı ($SERIAL). Cihazın aynı ağda olduğundan emin olun."
    exit 1
fi
# ---------------------------------------------------------------
# 3) BAGLANTI KURULDUKTAN SONRA EKRAN ORANI SORUSU (ZENITY --INFO + 2 BUTON)
# ---------------------------------------------------------------
# Bu asamada cihaz baglantisi hazir oldugundan kullanici kabloyu cikarabilir.

ZENITY_KOMUT="DISPLAY=$DISPLAY XAUTHORITY=$XAUTHORITY zenity --info \
    --title='$UYGULAMA_ADI' \
    --text='Tablet başarıyla bağlandı!\n\nKabloyu çıkarabilirsiniz.\nLütfen ekran görüntü oranını seçin:' \
    --ok-label='Tabletin Kendi Oranı' \
    --extra-button='Tahtada Tam Ekran (Kırpılır)' \
    --width=420 --height=180"

if [ -n "${GRAFIK_KULLANICI:-}" ]; then
    ZENITY_OUT="$(su - "$GRAFIK_KULLANICI" -c "$ZENITY_KOMUT" 2>/dev/null)"
    ZENITY_EXIT=$?
else
    ZENITY_OUT="$(eval "$ZENITY_KOMUT" 2>/dev/null)"
    ZENITY_EXIT=$?
fi

ORAN_SECIMI=1

if [ $ZENITY_EXIT -eq 0 ]; then
    # Ok label ("Tabletin Kendi Oranı") secildi
    ORAN_SECIMI=1
elif [ "$ZENITY_OUT" = "Tahtada Tam Ekran (Kırpılır)" ]; then
    # Ekstra buton secildi
    ORAN_SECIMI=2
else
    # Pencere kapatildi / iptal edildi
    exit 0
fi

# ---------------------------------------------------------------
# 4) EKRAN ORANI HESAPLAMASI
# ---------------------------------------------------------------
CROP_PARAM=()

if [ "$ORAN_SECIMI" = "2" ]; then
    SIZE_LINE="$(adb -s "$SERIAL" shell wm size 2>/dev/null | tail -n1)"
    RES="$(echo "$SIZE_LINE" | grep -oE '[0-9]+x[0-9]+')"

    if [ -n "$RES" ]; then
        WIDTH="${RES%x*}"
        HEIGHT="${RES#*x}"

        if [ "$WIDTH" -ge "$HEIGHT" ]; then
            ORAN_ASIM=$(awk -v w="$WIDTH" -v h="$HEIGHT" 'BEGIN{ print (w/h > 16/9) ? 1 : 0 }')
            if [ "$ORAN_ASIM" -eq 1 ]; then
                TARGET_H="$HEIGHT"
                TARGET_W=$(awk -v h="$HEIGHT" 'BEGIN{ printf "%d", h*16/9 }')
                [ "$TARGET_W" -gt "$WIDTH" ] && TARGET_W="$WIDTH"
                OFFSET_X=$(( (WIDTH - TARGET_W) / 2 ))
                OFFSET_Y=0
            else
                TARGET_W="$WIDTH"
                TARGET_H=$(awk -v w="$WIDTH" 'BEGIN{ printf "%d", w*9/16 }')
                [ "$TARGET_H" -gt "$HEIGHT" ] && TARGET_H="$HEIGHT"
                OFFSET_Y=$(( (HEIGHT - TARGET_H) / 2 ))
                OFFSET_X=0
            fi
        else
            ORAN_ASIM=$(awk -v w="$WIDTH" -v h="$HEIGHT" 'BEGIN{ print (h/w > 16/9) ? 1 : 0 }')
            if [ "$ORAN_ASIM" -eq 1 ]; then
                TARGET_W="$WIDTH"
                TARGET_H=$(awk -v w="$WIDTH" 'BEGIN{ printf "%d", w*16/9 }')
                [ "$TARGET_H" -gt "$HEIGHT" ] && TARGET_H="$HEIGHT"
                OFFSET_Y=$(( (HEIGHT - TARGET_H) / 2 ))
                OFFSET_X=0
            else
                TARGET_H="$HEIGHT"
                TARGET_W=$(awk -v h="$HEIGHT" 'BEGIN{ printf "%d", h*9/16 }')
                [ "$TARGET_W" -gt "$WIDTH" ] && TARGET_W="$WIDTH"
                OFFSET_X=$(( (WIDTH - TARGET_W) / 2 ))
                OFFSET_Y=0
            fi
        fi

        TARGET_W=$(( TARGET_W - (TARGET_W % 2) ))
        TARGET_H=$(( TARGET_H - (TARGET_H % 2) ))
        OFFSET_X=$(( OFFSET_X - (OFFSET_X % 2) ))
        OFFSET_Y=$(( OFFSET_Y - (OFFSET_Y % 2) ))

        CROP_VALUE="${TARGET_W}:${TARGET_H}:${OFFSET_X}:${OFFSET_Y}"
        CROP_PARAM=(--crop "$CROP_VALUE")
    fi
fi

# ---------------------------------------------------------------
# 5) SCRCPY BASLAT
# ---------------------------------------------------------------
scrcpy -s "$SERIAL" --max-size 1280 --video-bit-rate 2M --max-fps 30 -f "${CROP_PARAM[@]}" >/dev/null 2>&1

exit 0
